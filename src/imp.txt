#ifndef F_CPU
#define F_CPU 8000000UL
#endif

#include <avr/io.h>
#include <avr/eeprom.h>
#include <util/delay.h>
#include <string.h>

#define PASSWORD "1234"
#define GAME_DELAY 7

#define LCD_RS PA3
#define LCD_EN PA2

/*
    the number in keypad that we choose for password
*/
char keypadMap[4][4] =
{
    {'7','8','9','/'},
    {'4','5','6','*'},
    {'1','2','3','-'},
    {'C','0','=','+'}
};

/*
    the number in hexadecimal
*/

uint8_t segData[10] =
{
    0x3F,0x06,0x5B,0x4F,0x66,
    0x6D,0x7D,0x07,0x7F,0x6F
};
// EEMEM is function in AVR library that tell the chip to store in EEPROM not RAM
uint8_t EEMEM bestScore = 0;

void beep(int time)
{
    // بيخلي النقطه اللي بازر متصل بيها تبقي عاليه من غير ما يأثر عل الباقي
    PORTC |= (1<<PC6);

    /*
        the loop is for delay because if there is no llop the buzzer on and off so fast so loop to delay it to hear the buzzer
    */
    while(time--)
    {
        _delay_ms(1);
    }
    // بيخلي النقطه اللي بازر متصل بيها تبقي صفر من غير ما يأثر عل الباقي عشان يقفل البازر
    PORTC &= ~(1<<PC6);
}

void pulse()
{
    // هنا عشان تشغل الشاشه نفسها
    PORTA |= (1<<LCD_EN);
    /*
        هنا فيه 
        delay
        عشان من غيره الشاشه مش هتلحق تستقبل السيحنال ف مش هتعرض حاجه
    */
    _delay_us(100);
    PORTA &= ~(1<<LCD_EN);
}

void lcdWrite4(uint8_t value)
{
    PORTB &= 0x0F;          
    PORTB |= (value & 0xF0);
    pulse();
}

void lcdCmd(uint8_t value)
{
    PORTA &= ~(1<<LCD_RS);

    lcdWrite4(value);
    lcdWrite4(value << 4);

    if(value < 4)
        _delay_ms(2);
}

void lcdPut(char c)
{
    PORTA |= (1<<LCD_RS);

    lcdWrite4(c);
    lcdWrite4(c << 4);
}

void lcdPrint(char *text)
{
    while(*text)
        lcdPut(*text++);
}

void lcdGoto(uint8_t col, uint8_t row)
{
    if(row == 0)
        lcdCmd(0x80 + col);
    else
        lcdCmd(0xC0 + col);
}

void lcdBegin()
{
    DDRA |= (1<<LCD_RS) | (1<<LCD_EN) | (1<<PA4);
    DDRB = 0xFF;

    _delay_ms(50);

    lcdWrite4(0x30);
    _delay_ms(5);

    lcdWrite4(0x30);
    _delay_us(150);

    lcdWrite4(0x30);
    lcdWrite4(0x20);

    lcdCmd(0x28);
    lcdCmd(0x0C);
    lcdCmd(0x06);
    lcdCmd(0x01);
}

char getKey()
{
    DDRC |= 0x3C;
    PORTC |= 0x3C;

    DDRD &= ~((1<<PD3)|(1<<PD5)|(1<<PD6)|(1<<PD7));
    PORTD |= (1<<PD3)|(1<<PD5)|(1<<PD6)|(1<<PD7);

    int rowPins[4] = {PC5,PC4,PC3,PC2};
    int colPins[4] = {PD7,PD6,PD5,PD3};

    for(int r=0;r<4;r++)
    {
        PORTC &= ~(1<<rowPins[r]);

        _delay_us(20);

        for(int c=0;c<4;c++)
        {
            if(!(PIND & (1<<colPins[c])))
            {
                _delay_ms(30);

                while(!(PIND & (1<<colPins[c])));

                _delay_ms(30);

                return keypadMap[r][c];
            }
        }

        PORTC |= (1<<rowPins[r]);
    }

    return 0;
}

void showScore(uint8_t left,uint8_t right)
{
    PORTC &= ~((1<<PC6)|(1<<PC7));
    PORTB = segData[left % 10];
    PORTC |= (1<<PC7);

    _delay_ms(5);

    PORTC &= ~((1<<PC6)|(1<<PC7));
    PORTB = segData[right % 10];
    PORTC |= (1<<PC6);

    _delay_ms(5);
}

void checkPassword()
{
    char input[5]={0};
    int index=0;

    lcdCmd(0x01);
    lcdPrint("Enter Password");
    lcdGoto(0,1);

    while(1)
    {
        char key=getKey();

        if(!key)
            continue;

        if(key=='C')
        {
            index=0;

            lcdCmd(0x01);
            lcdPrint("Enter Password");
            lcdGoto(0,1);
        }
        else if(key>='0' && key<='9' && index<4)
        {
            input[index++]=key;
            lcdPut('*');
            beep(30);
        }

        if(index==4)
        {
            input[4]='\0';

            if(strcmp(input,PASSWORD)==0)
            {
                PORTA |= (1<<PA4);

                lcdCmd(0x01);
                lcdPrint("Correct");

                beep(100);

                _delay_ms(500);

                break;
            }

            DDRD |= (1<<PD2);
            PORTD |= (1<<PD2);

            lcdCmd(0x01);
            lcdPrint("Wrong");

            beep(200);

            _delay_ms(800);

            PORTD &= ~(1<<PD2);

            index=0;

            lcdCmd(0x01);
            lcdPrint("Enter Password");
            lcdGoto(0,1);
        }
    }
}

void drawGame(int ballX,int ballY,int leftPad,int rightPad)
{
    lcdCmd(0x01);

    lcdGoto(0,leftPad);
    lcdPut('|');

    lcdGoto(15,rightPad);
    lcdPut('|');

    if(ballX>=0 && ballX<=15)
    {
        lcdGoto(ballX,ballY);
        lcdPut('O');
    }
}

int main()
{
    DDRB=0xFF;
    DDRC=0xFF;
    DDRD=0x04;

    lcdBegin();

    checkPassword();

    uint8_t high=eeprom_read_byte(&bestScore);

    if(high>99)
        high=0;

    int leftScore=0;
    int rightScore=0;

    int ballX=7;
    int ballY=0;
    int dir=1;

    while(1)
    {
        PORTC &= ~(1<<PC5);

        _delay_us(10);

        int leftPlayer = !(PIND & (1<<PD7));
        int rightPlayer = !(PIND & (1<<PD3));

        PORTC |= (1<<PC5);

        ballX += dir;
        ballY = ballX % 2;

        if(ballX==1 && ballY==leftPlayer)
            dir=1;

        if(ballX==14 && ballY==rightPlayer)
            dir=-1;

        if(ballX<0 || ballX>15)
        {
            if(ballX<0)
                rightScore++;
            else
                leftScore++;

            beep(100);

            ballX=7;
            dir=-dir;

            if(leftScore>high || rightScore>high)
            {
                high = (leftScore>rightScore) ? leftScore : rightScore;
                eeprom_update_byte(&bestScore,high);
            }
        }

        drawGame(ballX,ballY,leftPlayer,rightPlayer);

        for(int i=0;i<GAME_DELAY;i++)
            showScore(leftScore,rightScore);
    }
}


















#ifndef F_CPU
#define F_CPU 8000000UL
#endif

#include <avr/io.h>
#include <util/delay.h>
#include <avr/eeprom.h>
#include <string.h>
#include <stdlib.h>

// --- PIN DEFINITIONS (32 Mini Board) ---
#define LED_RED     PC0
#define LED_GREEN   PC1
#define BUZZER      PC5
#define DIG_LEFT    PC6
#define DIG_RIGHT   PC7
#define LCD_RS      PA1
#define LCD_EN      PA2

// 7-Seg mapping: A=PA1, B=PA2, C=PA3, D=PA4, E=PA5, F=PA6, G=PA7
const uint8_t segChars[] = {0x7E, 0x0C, 0xB6, 0x9E, 0xCC, 0xDA, 0xFA, 0x0E, 0xFE, 0xDE};
char keypadMap[4][4] = {{'7','8','9','/'}, {'4','5','6','*'}, {'1','2','3','-'}, {'C','0','=','+'}};

uint8_t EEMEM nvHighScore = 0;
unsigned int seedCounter = 0;

// --- UTILITIES ---
void beep(int ms) {
    PORTC |= (1<<BUZZER);
    for(int i=0; i<ms; i++) _delay_ms(1);
    PORTC &= ~(1<<BUZZER);
}

// --- LCD FUNCTIONS (Multiplexed Port A) ---
void lcd_pulse() {
    PORTA |= (1<<LCD_EN); _delay_us(10);
    PORTA &= ~(1<<LCD_EN); _delay_us(10);
}

void lcd_write4(uint8_t val) {
    uint8_t rs_state = PORTA & 0x02; 
    PORTA = rs_state | ((val & 0xF0) >> 1);
    lcd_pulse();
}

void lcd_cmd(uint8_t cmd) {
    PORTC &= ~((1<<DIG_LEFT) | (1<<DIG_RIGHT));
    PORTA &= ~(1<<LCD_RS);
    lcd_write4(cmd); lcd_write4(cmd << 4);
    _delay_ms(1);
}

void lcd_char(char c) {
    PORTC &= ~((1<<DIG_LEFT) | (1<<DIG_RIGHT));
    PORTA |= (1<<LCD_RS);
    lcd_write4(c); lcd_write4(c << 4);
    _delay_us(40);
}

void lcd_print(char *str) { while(*str) lcd_char(*str++); }

void lcd_init() {
    DDRA = 0xFE; DDRC |= 0xE3; DDRB = 0xFF;
    _delay_ms(20);
    lcd_write4(0x30); _delay_ms(5);
    lcd_write4(0x30); _delay_us(150);
    lcd_write4(0x30); lcd_write4(0x20); 
    lcd_cmd(0x28); lcd_cmd(0x0C); lcd_cmd(0x01);
}

// --- KEYPAD ---
char getKey() {
    DDRB |= 0xF0; PORTB |= 0xF0;
    DDRD &= ~0x3C; PORTD |= 0x3C;
    int rPins[] = {PB4, PB5, PB6, PB7}, cPins[] = {PD2, PD3, PD4, PD5};
    for(int r=0; r<4; r++) {
        PORTB &= ~(1<<rPins[r]); _delay_us(5);
        for(int c=0; c<4; c++) {
            if(!(PIND & (1<<cPins[c]))) {
                _delay_ms(20); while(!(PIND & (1<<cPins[c])));
                return keypadMap[r][c];
            }
        }
        PORTB |= (1<<rPins[r]);
    }
    return 0;
}

// --- SCORE DISPLAY (7-SEG) ---
void showScore(uint8_t s1, uint8_t s2) {
    PORTA &= ~((1<<LCD_RS) | (1<<LCD_EN)); // Ensure LCD pins are quiet
    
    PORTA = segChars[s1 % 10]; 
    PORTC |= (1<<DIG_LEFT); _delay_ms(4); PORTC &= ~(1<<DIG_LEFT);
    
    PORTA = segChars[s2 % 10]; 
    PORTC |= (1<<DIG_RIGHT); _delay_ms(4); PORTC &= ~(1<<DIG_RIGHT);
}

// --- PASSWORD & STARTUP ---
void startSequence() {
    char pass[] = "1234", input[5] = {0};
    int idx = 0;
    uint8_t hScore = eeprom_read_byte(&nvHighScore);
    if(hScore > 99) hScore = 0;

    while(1) {
        lcd_cmd(0x01); lcd_print("Enter Password");
        lcd_cmd(0xC0); idx = 0;
        while(idx < 4) {
            seedCounter++; 
            char k = getKey();
            if(k >= '0' && k <= '9') { input[idx++] = k; lcd_char('*'); beep(20); }
        }
        if(strcmp(input, pass) == 0) {
            srand(seedCounter);
            PORTC |= (1<<LED_GREEN); PORTC &= ~(1<<LED_RED);
            lcd_cmd(0x01); lcd_print("Correct!"); _delay_ms(500);
            
            // SHOW HIGH SCORE DURING LOADING
            lcd_cmd(0x01); lcd_print("High Score: ");
            lcd_char((hScore/10)+'0'); lcd_char((hScore%10)+'0');
            lcd_cmd(0xC0); lcd_print("Loading Game...");
            _delay_ms(1500); 
            return;
        } else {
            PORTC |= (1<<LED_RED); PORTC &= ~(1<<LED_GREEN);
            lcd_cmd(0x01); lcd_print("Wrong Pass!"); beep(200); _delay_ms(800);
        }
    }
}

int main() {
    lcd_init();
    startSequence();

    uint8_t currentHighScore = eeprom_read_byte(&nvHighScore);
    if(currentHighScore > 99) currentHighScore = 0;

    int ballX = 7, ballY = 0, dx = 1, dy = 1;
    int p1Y = 0, p2Y = 0, s1 = 0, s2 = 0;
    char line1[17], line2[17];

    while(1) {
        char k = getKey();
        if(k == '7') p1Y = 0; if(k == '1') p1Y = 1;
        if(k == '9') p2Y = 0; if(k == '3') p2Y = 1;

        ballX += dx; ballY += dy;
        if(ballY < 0) { ballY = 0; dy = 1; } 
        if(ballY > 1) { ballY = 1; dy = -1; }

        // --- POINT LOST LOGIC ---
        if(ballX < 0 || ballX > 15) {
            if(ballX < 0) s2++; else s1++; // Record points
            
            beep(400);
            lcd_cmd(0x01); lcd_print("  GAME OVER!");
            lcd_cmd(0xC0); // Show current scores of the round
            lcd_print("P1:"); lcd_char(s1+'0'); lcd_print("  P2:"); lcd_char(s2+'0');
            
            // EEPROM Record Update
            uint8_t winnerScore = (s1 > s2) ? s1 : s2;
            if(winnerScore > currentHighScore) {
                currentHighScore = winnerScore;
                eeprom_update_byte(&nvHighScore, currentHighScore);
            }
            
            // Hold display so user sees the score
            for(int j=0; j<150; j++) showScore(s1, s2);
            
            // Reset game but keep EEPROM updated
            s1 = s2 = 0; ballX = 7; 
            dx = (rand() % 2 == 0) ? 1 : -1;
            dy = (rand() % 2 == 0) ? 1 : -1;
        }

        // Paddle Collision
        if(ballX == 0 && ballY == p1Y) { dx = 1; beep(10); }
        if(ballX == 15 && ballY == p2Y) { dx = -1; beep(10); }

        // Render Strings
        memset(line1, ' ', 16); memset(line2, ' ', 16);
        line1[16] = line2[16] = '\0';
        if(p1Y == 0) line1[0] = '|'; else line2[0] = '|';
        if(p2Y == 0) line1[15] = '|'; else line2[15] = '|';
        if(ballX >= 0 && ballX <= 15) {
            if(ballY == 0) line1[ballX] = 'O'; else line2[ballX] = 'O';
        }

        lcd_cmd(0x80); lcd_print(line1);
        lcd_cmd(0xC0); lcd_print(line2);
        
        // Show Score on 7-Seg
        for(int j=0; j<8; j++) showScore(s1, s2);
    }
}

// --- PIN DEFINITIONS (32 Mini Board) ---
#define LED_RED     PC0

#define BUZZER      PC5

#define DIG_RIGHT   PC7
#define LCD_RS      PA1
#define LCD_EN      PA2

// 7-Seg mapping: A=PA1, B=PA2, C=PA3, D=PA4, E=PA5, F=PA6, G=PA7
const uint8_t segChars[] = {0x7E, 0x0C, 0xB6, 0x9E, 0xCC, 0xDA, 0xFA, 0x0E, 0xFE, 0xDE};
char keypadMap[4][4] = {{'7','8','9','/'}, {'4','5','6','*'}, {'1','2','3','-'}, {'C','0','=','+'}};

// EEPROM storage for the pair (Record P1 and Record P2)
uint8_t EEMEM nvHighP1 = 0;
uint8_t EEMEM nvHighP2 = 0;
unsigned int seedCounter = 0;

// --- UTILITIES ---
void beep(int ms) {
    PORTC |= (1<<BUZZER);
    for(int i=0; i<ms; i++) _delay_ms(1);
    PORTC &= ~(1<<BUZZER);
}

// --- LCD FUNCTIONS ---
void lcd_pulse() {
    PORTA |= (1<<LCD_EN); _delay_us(10);
    PORTA &= ~(1<<LCD_EN); _delay_us(10);
}

void lcd_write4(uint8_t val) {
    uint8_t rs_state = PORTA & 0x02; 
    PORTA = rs_state | ((val & 0xF0) >> 1);
    lcd_pulse();
}

    PORTC &= ~((1<<DIG_LEFT) | (1<<DIG_RIGHT));
    PORTA &= ~(1<<LCD_RS);
    lcd_write4(cmd); lcd_write4(cmd << 4);
    _delay_ms(2); // Proteus stable delay
}

void lcd_char(char c) {
    PORTC &= ~((1<<DIG_LEFT) | (1<<DIG_RIGHT));
    PORTA |= (1<<LCD_RS);
    lcd_write4(c); lcd_write4(c << 4);
    _delay_us(50);
}

void lcd_print(char *str) { while(*str) lcd_char(*str++); }

void lcd_init() {
    DDRA = 0xFE; DDRC |= 0xE3; DDRB = 0xFF;
    _delay_ms(20);
    lcd_write4(0x30); _delay_ms(5);
    lcd_write4(0x30); _delay_us(150);
    lcd_write4(0x30); lcd_write4(0x20); 
    lcd_cmd(0x28); lcd_cmd(0x0C); lcd_cmd(0x01);
}

char getKey() {
    DDRB |= 0xF0; PORTB |= 0xF0;
    DDRD &= ~0x3C; PORTD |= 0x3C;
    int rPins[] = {PB4, PB5, PB6, PB7}, cPins[] = {PD2, PD3, PD4, PD5};
    for(int r=0; r<4; r++) {
        PORTB &= ~(1<<rPins[r]); _delay_us(5);
        for(int c=0; c<4; c++) {
            if(!(PIND & (1<<cPins[c]))) {
                _delay_ms(20); while(!(PIND & (1<<cPins[c])));
                return keypadMap[r][c];
            }
        }
        PORTB |= (1<<rPins[r]);
    }

}

void showScore(uint8_t s1, uint8_t s2) {
    PORTA &= ~((1<<LCD_RS) | (1<<LCD_EN)); 
    PORTA = segChars[s1 % 10]; PORTC |= (1<<DIG_LEFT); _delay_ms(4); PORTC &= ~(1<<DIG_LEFT);
    PORTA = segChars[s2 % 10]; PORTC |= (1<<DIG_RIGHT); _delay_ms(4); PORTC &= ~(1<<DIG_RIGHT);
}

void startSequence() {
    char pass[] = "1234", input[5] = {0};
    int idx = 0;
    while(1) {
        lcd_cmd(0x01); lcd_print("Enter Password");
        lcd_cmd(0xC0); idx = 0;
        while(idx < 4) { seedCounter++; char k = getKey(); if(k >= '0' && k <= '9') { input[idx++] = k; lcd_char('*'); beep(20); } }
        if(strcmp(input, pass) == 0) {
            srand(seedCounter);
            PORTC |= (1<<LED_GREEN); PORTC &= ~(1<<LED_RED);
            lcd_cmd(0x01); lcd_print("Correct!"); _delay_ms(500);
            
            // LOAD RECORD AS PAIR
            uint8_t hp1 = eeprom_read_byte(&nvHighP1);
            uint8_t hp2 = eeprom_read_byte(&nvHighP2);
            if(hp1 > 99) hp1 = 0; if(hp2 > 99) hp2 = 0;
            
            lcd_cmd(0x01); lcd_print("Record: "); 
            lcd_char((hp1/10)+'0'); lcd_char((hp1%10)+'0');
            lcd_char(':');
            lcd_char((hp2/10)+'0'); lcd_char((hp2%10)+'0');
            
            lcd_cmd(0xC0); lcd_print("Loading Game..."); _delay_ms(1500); return;
        } else {
            PORTC |= (1<<LED_RED); PORTC &= ~(1<<LED_GREEN);
            lcd_cmd(0x01); lcd_print("Wrong Pass!"); beep(200); _delay_ms(800);
        }
    }
}

int main() {
    lcd_init();
    startSequence();

    int ballX = 7, ballY = 0, dx = 1, dy = 1;
    int p1Y = 0, p2Y = 0, s1 = 0, s2 = 0;
    char line1[17], line2[17];

    while(1) {
        char k = getKey();
        if(k == '7') p1Y = 0; if(k == '1') p1Y = 1;
        if(k == '9') p2Y = 0; if(k == '3') p2Y = 1;

        // --- STOP SESSION (BUTTON 5) ---
        if(k == '5') {
            beep(600);
            lcd_cmd(0x01); lcd_print("SESSION ENDED");
            lcd_cmd(0xC0); 
            lcd_print("Final: "); lcd_char((s1/10)+'0'); lcd_char((s1%10)+'0');
            lcd_char(':'); lcd_char((s2/10)+'0'); lcd_char((s2%10)+'0');
            
            // Check if this session's total beat the record total
            uint8_t oldP1 = eeprom_read_byte(&nvHighP1);
            uint8_t oldP2 = eeprom_read_byte(&nvHighP2);
            if((s1 + s2) > (oldP1 + oldP2)) {
                eeprom_update_byte(&nvHighP1, s1);
                eeprom_update_byte(&nvHighP2, s2);
                _delay_ms(1000); lcd_cmd(0x01); lcd_print(" NEW RECORD! ");
            }
            for(int j=0; j<250; j++) showScore(s1, s2);
            s1 = s2 = 0; ballX = 7; 
            lcd_cmd(0x01); lcd_print("Restarting..."); _delay_ms(1000);
            continue;
        }

        ballX += dx; ballY += dy;
        if(ballY < 0) { ballY = 0; dy = 1; } if(ballY > 1) { ballY = 1; dy = -1; }

        // --- WINNER LOGIC (Fixing Glitch) ---
        if(ballX < 0 || ballX > 15) {
            if(ballX < 0) s2++; else s1++;
            beep(300);
            lcd_cmd(0x01); _delay_ms(5);
            lcd_print("  ROUND OVER!");
            lcd_cmd(0xC0); lcd_print("Score: "); 
            lcd_char((s1/10)+'0'); lcd_char((s1%10)+'0');
            lcd_print(":");
            lcd_char((s2/10)+'0'); lcd_char((s2%10)+'0');
            
            for(int j=0; j<150; j++) showScore(s1, s2);
            ballX = 7; ballY = 0; // Reset ball to middle
            dx = (rand() % 2 == 0) ? 1 : -1;
            dy = (rand() % 2 == 0) ? 1 : -1;
            continue; // CRITICAL: Stop drawing this frame to avoid glitch
        }

        if(ballX == 0 && ballY == p1Y) { dx = 1; beep(5); }
        if(ballX == 15 && ballY == p2Y) { dx = -1; beep(5); }

        // --- ANTI-GLITCH RENDERING ---
        memset(line1, ' ', 16); memset(line2, ' ', 16);
        line1[16] = line2[16] = '\0';
        
        if(p1Y == 0) line1[0] = '|'; else line2[0] = '|';
        if(p2Y == 0) line1[15] = '|'; else line2[15] = '|';
        
        // BUFFER PROTECTION: Only draw if X is 0 to 15
        if(ballX >= 0 && ballX < 16) {
            if(ballY == 0) line1[ballX] = 'O'; else line2[ballX] = 'O';
        }

static const char KEYPAD_MAP[4][4] = {
    {'7', '8', '9', '/'},
    {'4', '5', '6', '*'},
    {'1', '2', '3', '-'},
    {'C', '0', '=', '+'}
};

uint8_t EEMEM nvHighP1 = 0;
uint8_t EEMEM nvHighP2 = 0;

static unsigned int g_seed_counter = 0;

void buzzer_beep(uint16_t duration_ms) {
    PORTC |= (1 << BUZZER);
    while (duration_ms--) {
        _delay_ms(1);
    }
    PORTC &= ~(1 << BUZZER);
}

static inline void lcd_pulse_enable(void) {
    PORTA |= (1 << LCD_EN);   _delay_us(10);
    PORTA &= ~(1 << LCD_EN);  _delay_us(10);
}

void lcd_write_nibble(uint8_t nibble) {
    uint8_t current_rs = PORTA & 0x02; 
    PORTA = current_rs | ((nibble & 0xF0) >> 1);
    lcd_pulse_enable();
}

void lcd_send_cmd(uint8_t cmd) {
    PORTC &= ~((1 << DIG_LEFT) | (1 << DIG_RIGHT));
    PORTA &= ~(1 << LCD_RS);
    lcd_write_nibble(cmd); 
    lcd_write_nibble(cmd << 4);
    _delay_ms(2); 
}

void lcd_send_char(char c) {
    PORTC &= ~((1 << DIG_LEFT) | (1 << DIG_RIGHT));
    PORTA |= (1 << LCD_RS);
    lcd_write_nibble(c); 
    lcd_write_nibble(c << 4);
    _delay_us(50);
}

void lcd_print_string(const char *str) {
    while (*str) {
        lcd_send_char(*str++);
    }
}

void lcd_hardware_init(void) {
    DDRA  = 0xFE; 
    DDRC |= 0xE3; 
    DDRB  = 0xFF;
    
    _delay_ms(20);
    lcd_write_nibble(0x30); _delay_ms(5);
    lcd_write_nibble(0x30); _delay_us(150);
    lcd_write_nibble(0x30); lcd_write_nibble(0x20); 
    
    lcd_send_cmd(0x28); 
    lcd_send_cmd(0x0C); 
    lcd_send_cmd(0x01);
}

char keypad_scan(void) {
    DDRB |= 0xF0;  PORTB |= 0xF0;
    DDRD &= ~0x3C; PORTD |= 0x3C;
    
    const uint8_t row_pins[] = {PB4, PB5, PB6, PB7};
    const uint8_t col_pins[] = {PD2, PD3, PD4, PD5};
    
    for (uint8_t r = 0; r < 4; r++) {
        PORTB &= ~(1 << row_pins[r]); 
        _delay_us(5);
        
        for (uint8_t c = 0; c < 4; c++) {
            if (!(PIND & (1 << col_pins[c]))) {
                _delay_ms(20);
                while (!(PIND & (1 << col_pins[c])));
                return KEYPAD_MAP[r][c];
            }
        }
        PORTB |= (1 << row_pins[r]);
    }
    return 0;
}

void display_refresh_score(uint8_t score1, uint8_t score2) {
    uint8_t lcd_ctrl_bits = PORTA & ((1 << LCD_RS) | (1 << LCD_EN));
    uint8_t pa0_state     = PORTA & 0x01;

    uint8_t left_pattern  = SEGM_CHARS[score1 % 10] & ~((1 << LCD_RS) | (1 << LCD_EN));
    uint8_t right_pattern = SEGM_CHARS[score2 % 10] & ~((1 << LCD_RS) | (1 << LCD_EN));

    PORTA = lcd_ctrl_bits | left_pattern | pa0_state;
    PORTC |= (1 << DIG_LEFT); 
    _delay_ms(4); 
    PORTC &= ~(1 << DIG_LEFT);
    
    PORTA = lcd_ctrl_bits | right_pattern | pa0_state;
    PORTC |= (1 << DIG_RIGHT); 
    _delay_ms(4); 
    PORTC &= ~(1 << DIG_RIGHT);
}

void run_authentication_sequence(void) {
    const char correct_password[] = "1234";
    char input_buffer[5] = {0};
    uint8_t input_index = 0;
    
    while (1) {
        lcd_send_cmd(0x01); 
        lcd_print_string("Enter Password");
        lcd_send_cmd(0xC0); 
        input_index = 0;
        
        while (input_index < 4) {
            g_seed_counter++; 
            char key = keypad_scan(); 
            if (key >= '0' && key <= '9') { 
                input_buffer[input_index++] = key; 
                lcd_send_char('*'); 
                buzzer_beep(20); 
            } 
        }
        
        if (strcmp(input_buffer, correct_password) == 0) {
            srand(g_seed_counter);
            PORTC |= (1 << LED_GREEN); 
            PORTC &= ~(1 << LED_RED);
            
            lcd_send_cmd(0x01); 
            lcd_print_string("Correct!"); 
            _delay_ms(500);
            
            uint8_t high_score_p1 = eeprom_read_byte(&nvHighP1);
            uint8_t high_score_p2 = eeprom_read_byte(&nvHighP2);
            if (high_score_p1 > 99) high_score_p1 = 0; 
            if (high_score_p2 > 99) high_score_p2 = 0;
            
            lcd_send_cmd(0x01); 
            lcd_print_string("Record: "); 
            lcd_send_char((high_score_p1 / 10) + '0'); 
            lcd_send_char((high_score_p1 % 10) + '0');
            lcd_send_char(':');
            lcd_send_char((high_score_p2 / 10) + '0'); 
            lcd_send_char((high_score_p2 % 10) + '0');
            
            lcd_send_cmd(0xC0); 
            lcd_print_string("Loading Game..."); 
            _delay_ms(1500); 
            return;
        } else {
            PORTC |= (1 << LED_RED); 
            PORTC &= ~(1 << LED_GREEN);
            lcd_send_cmd(0x01); 
            lcd_print_string("Wrong Pass!"); 
            buzzer_beep(200); 
            _delay_ms(800);
        }
    }
}

int main(void) {
    lcd_hardware_init();
    run_authentication_sequence();

    int8_t ball_x = 7, ball_y = 0;
    int8_t velocity_x = 1, velocity_y = 1;
    uint8_t paddle_p1_y = 0, paddle_p2_y = 0;
    uint8_t score_p1 = 0, score_p2 = 0;
    
    char render_buffer_l1[17];
    char render_buffer_l2[17];

    while (1) {
        char action_key = keypad_scan();
        
        if (action_key == '7') paddle_p1_y = 0; 
        if (action_key == '1') paddle_p1_y = 1;
        if (action_key == '9') paddle_p2_y = 0; 
        if (action_key == '3') paddle_p2_y = 1;

        if (action_key == '5') {
            buzzer_beep(600);
            lcd_send_cmd(0x01); 
            lcd_print_string("SESSION ENDED");
            lcd_send_cmd(0xC0); 
            lcd_print_string("Final: "); 
            lcd_send_char((score_p1 / 10) + '0');  lcd_send_char((score_p1 % 10) + '0');
            lcd_send_char(':'); 
            lcd_send_char((score_p2 / 10) + '0');  lcd_send_char((score_p2 % 10) + '0');
            
            uint8_t stored_p1 = eeprom_read_byte(&nvHighP1);
            uint8_t stored_p2 = eeprom_read_byte(&nvHighP2);
            if (stored_p1 > 99) stored_p1 = 0;
            if (stored_p2 > 99) stored_p2 = 0;
            
            if ((score_p1 > stored_p1) || (score_p1 == stored_p1 && score_p2 > stored_p2)) {
                eeprom_update_byte(&nvHighP1, score_p1);
                eeprom_update_byte(&nvHighP2, score_p2);
                _delay_ms(1000); 
                lcd_send_cmd(0x01); 
                lcd_print_string(" NEW RECORD! ");
            }
            
            for (uint8_t cycles = 0; cycles < 250; cycles++) {
                display_refresh_score(score_p1, score_p2);
            }
            
            score_p1 = 0; 
            score_p2 = 0; 
            ball_x = 7; 
            
            lcd_hardware_init(); 
            lcd_print_string("Restarting..."); 
            _delay_ms(1000);
            continue;
        }

        ball_x += velocity_x; 
        ball_y += velocity_y;
        
        if (ball_y < 0) { ball_y = 0; velocity_y = 1; } 
        if (ball_y > 1) { ball_y = 1; velocity_y = -1; }

        if (ball_x < 0 || ball_x > 15) {
            if (ball_x < 0) score_p2++; else score_p1++;
            buzzer_beep(300);
            
            lcd_hardware_init(); 
            lcd_send_cmd(0x01);
            lcd_print_string("   ROUND OVER!");
            lcd_send_cmd(0xC0); 
            lcd_print_string("Score: "); 
            lcd_send_char((score_p1 / 10) + '0');  lcd_send_char((score_p1 % 10) + '0');
            lcd_print_string(":");
            lcd_send_char((score_p2 / 10) + '0');  lcd_send_char((score_p2 % 10) + '0');
            
            for (uint8_t cycles = 0; cycles < 150; cycles++) {
                display_refresh_score(score_p1, score_p2);
            }
            
            ball_x = 7; 
            ball_y = 0; 
            velocity_x = (rand() % 2 == 0) ? 1 : -1;
            velocity_y = (rand() % 2 == 0) ? 1 : -1;
            
            lcd_hardware_init(); 
            continue; 
        }

        if (ball_x == 0 && ball_y == paddle_p1_y)  { velocity_x = 1;  buzzer_beep(5); }
        if (ball_x == 15 && ball_y == paddle_p2_y) { velocity_x = -1; buzzer_beep(5); }

        memset(render_buffer_l1, ' ', 16); 
        memset(render_buffer_l2, ' ', 16);
        render_buffer_l1[16] = render_buffer_l2[16] = '\0';
        
        if (paddle_p1_y == 0) render_buffer_l1[0]  = '|'; else render_buffer_l2[0]  = '|';
        if (paddle_p2_y == 0) render_buffer_l1[15] = '|'; else render_buffer_l2[15] = '|';
        
        if (ball_x >= 0 && ball_x < 16) {
            if (ball_y == 0) render_buffer_l1[ball_x] = 'O'; else render_buffer_l2[ball_x] = 'O';
        }

        lcd_send_cmd(0x80); lcd_print_string(render_buffer_l1);
        lcd_send_cmd(0xC0); lcd_print_string(render_buffer_l2);
        
        for (uint8_t cycles = 0; cycles < 8; cycles++) {
            display_refresh_score(score_p1, score_p2);
        }
    }
}

    while (duration_ms--) {
        _delay_ms(1);
    }
    PORTC &= ~(1 << BUZZER);
}

static inline void lcd_pulse_enable(void) {
    PORTA |= (1 << LCD_EN);   _delay_us(10);
    PORTA &= ~(1 << LCD_EN);  _delay_us(10);
}

void lcd_write_nibble(uint8_t nibble) {
    uint8_t current_rs = PORTA & 0x02; 
    PORTA = current_rs | ((nibble & 0xF0) >> 1);
    lcd_pulse_enable();
}

void lcd_send_cmd(uint8_t cmd) {
    PORTC &= ~((1 << DIG_LEFT) | (1 << DIG_RIGHT));
    PORTA &= ~(1 << LCD_RS);
    lcd_write_nibble(cmd); 
    lcd_write_nibble(cmd << 4);
    _delay_ms(2); 
}

void lcd_send_char(char c) {
    PORTC &= ~((1 << DIG_LEFT) | (1 << DIG_RIGHT));
    PORTA |= (1 << LCD_RS);
    lcd_write_nibble(c); 
    lcd_write_nibble(c << 4);
    _delay_us(50);
}

void lcd_print_string(const char *str) {
    while (*str) {
        lcd_send_char(*str++);
    }
}

void lcd_hardware_init(void) {
    DDRA  = 0xFE; 
    DDRC |= 0xE3; 
    DDRB  = 0xFF;
    
    _delay_ms(50); 
    
    lcd_write_nibble(0x30); 
    _delay_ms(5);
    lcd_write_nibble(0x30); 
    _delay_us(200);
    lcd_write_nibble(0x30); 
    _delay_us(200);
    
    lcd_write_nibble(0x20); 
    _delay_ms(2);
    
    lcd_send_cmd(0x28); 
    lcd_send_cmd(0x0C); 
    lcd_send_cmd(0x01);
    _delay_ms(2);
}

char keypad_scan(void) {
    DDRB |= 0xF0;  PORTB |= 0xF0;
    DDRD &= ~0x3C; PORTD |= 0x3C;
    
    const uint8_t row_pins[] = {PB4, PB5, PB6, PB7};
    const uint8_t col_pins[] = {PD2, PD3, PD4, PD5};
    
    for (uint8_t r = 0; r < 4; r++) {
        PORTB &= ~(1 << row_pins[r]); 
        _delay_us(5);
        
        for (uint8_t c = 0; c < 4; c++) {
            if (!(PIND & (1 << col_pins[c]))) {
                _delay_ms(20);
                while (!(PIND & (1 << col_pins[c])));
                return KEYPAD_MAP[r][c];
            }
        }
        PORTB |= (1 << row_pins[r]);
    }
    return 0;
}

void display_refresh_score(uint8_t score1, uint8_t score2) {
    uint8_t lcd_ctrl_bits = PORTA & 0x01; 

    uint8_t left_pattern  = SEGM_CHARS[score1 % 10];
    uint8_t right_pattern = SEGM_CHARS[score2 % 10];

    PORTA = lcd_ctrl_bits | left_pattern; 
    PORTC |= (1 << DIG_LEFT);               
    _delay_ms(2);                           
    PORTC &= ~(1 << DIG_LEFT);              
    
    PORTA = lcd_ctrl_bits | right_pattern; 
    PORTC |= (1 << DIG_RIGHT);              
    _delay_ms(2);                           
    PORTC &= ~(1 << DIG_RIGHT);             
    
    PORTA &= 0x01;
}

void run_authentication_sequence(void) {
    const char correct_password[] = "1234";
    char input_buffer[5] = {0};
    uint8_t input_index = 0;
    
    while (1) {
        lcd_send_cmd(0x01); 
        lcd_print_string("Enter Password");
        lcd_send_cmd(0xC0); 
        input_index = 0;
        
        while (input_index < 4) {
            g_seed_counter++; 
            char key = keypad_scan(); 
            if (key >= '0' && key <= '9') { 
                input_buffer[input_index++] = key; 
                lcd_send_char('*'); 
                buzzer_beep(20); 
            } 
        }
        
        if (strcmp(input_buffer, correct_password) == 0) {
            srand(g_seed_counter);
            PORTC |= (1 << LED_GREEN); 
            PORTC &= ~(1 << LED_RED);
            
            lcd_send_cmd(0x01); 
            lcd_print_string("Correct!"); 
            _delay_ms(500);
            
            uint8_t high_score_p1 = eeprom_read_byte(&nvHighP1);
            uint8_t high_score_p2 = eeprom_read_byte(&nvHighP2);
            if (high_score_p1 > 99) high_score_p1 = 0; 
            if (high_score_p2 > 99) high_score_p2 = 0;
            
            lcd_send_cmd(0x01); 
            lcd_print_string("Record: "); 
            lcd_send_char((high_score_p1 / 10) + '0'); 
            lcd_send_char((high_score_p1 % 10) + '0');
            lcd_send_char(':');
            lcd_send_char((high_score_p2 / 10) + '0'); 
            lcd_send_char((high_score_p2 % 10) + '0');
            
            lcd_send_cmd(0xC0); 
            lcd_print_string("Loading Game..."); 
            _delay_ms(1500); 
            return;
        } else {
            PORTC |= (1 << LED_RED); 
            PORTC &= ~(1 << LED_GREEN);
            lcd_send_cmd(0x01); 
            lcd_print_string("Wrong Pass!"); 
            buzzer_beep(200); 
            _delay_ms(800);
        }
    }
}

int main(void) 
{
    lcd_hardware_init();
    run_authentication_sequence();

    int8_t ball_x = 7, ball_y = 0;
    int8_t velocity_x = 1, velocity_y = 1;
    uint8_t paddle_p1_y = 0, paddle_p2_y = 0;
    uint8_t score_p1 = 0, score_p2 = 0;
    
    char render_buffer_l1[17];
    char render_buffer_l2[17];

    while (1) {
        char action_key = keypad_scan();
        
        if (action_key == '7') paddle_p1_y = 0; 
        if (action_key == '1') paddle_p1_y = 1;
        if (action_key == '9') paddle_p2_y = 0; 
        if (action_key == '3') paddle_p2_y = 1;

        if (action_key == '5') {
            buzzer_beep(600);
            lcd_send_cmd(0x01); 
            lcd_print_string("SESSION ENDED");
            lcd_send_cmd(0xC0); 
            lcd_print_string("Final: "); 
            lcd_send_char((score_p1 / 10) + '0');  lcd_send_char((score_p1 % 10) + '0');
            lcd_send_char(':'); 
            lcd_send_char((score_p2 / 10) + '0');  lcd_send_char((score_p2 % 10) + '0');
            
            uint8_t stored_p1 = eeprom_read_byte(&nvHighP1);
            uint8_t stored_p2 = eeprom_read_byte(&nvHighP2);
            if (stored_p1 > 99) stored_p1 = 0;
            if (stored_p2 > 99) stored_p2 = 0;
            
            if ((score_p1 > stored_p1) || (score_p1 == stored_p1 && score_p2 > stored_p2)) {
                eeprom_update_byte(&nvHighP1, score_p1);
                eeprom_update_byte(&nvHighP2, score_p2);
                _delay_ms(1000); 
                lcd_send_cmd(0x01); 
                lcd_print_string(" NEW RECORD! ");
            }
            
            for (uint8_t cycles = 0; cycles < 250; cycles++) {
                display_refresh_score(score_p1, score_p2);
            }
            
            score_p1 = 0; 
            score_p2 = 0; 
            ball_x = 7; 
            
            lcd_hardware_init(); 
            lcd_print_string("Restarting..."); 
            _delay_ms(1000);
            continue;
        }

        ball_x += velocity_x; 
        ball_y += velocity_y;
        
        if (ball_y < 0) { ball_y = 0; velocity_y = 1; } 
        if (ball_y > 1) { ball_y = 1; velocity_y = -1; }

        if (ball_x < 0 || ball_x > 15) {
            if (ball_x < 0) score_p2++; else score_p1++;
            buzzer_beep(300);
            
            lcd_send_cmd(0x01); 
            lcd_print_string("   ROUND OVER!");
            lcd_send_cmd(0xC0); 
            lcd_print_string("Score: "); 
            lcd_send_char(score_p1 + '0'); 
            lcd_print_string(":");
            lcd_send_char(score_p2 + '0');
            
            PORTA &= ~((1 << LCD_RS) | (1 << LCD_EN));
            PORTC &= ~((1 << DIG_LEFT) | (1 << DIG_RIGHT));

            for (uint8_t cycles = 0; cycles < 200; cycles++) {
                display_refresh_score(score_p1, score_p2);
            }
            
            lcd_hardware_init(); 
            lcd_send_cmd(0x01);
            _delay_ms(5);

            if (score_p1 >= 10 || score_p2 >= 10) {
                buzzer_beep(800); 
                lcd_send_cmd(0x01);
                
                if (score_p1 >= 10) {
                    lcd_print_string("  PLAYER 1 WINS!");
                } else {
                    lcd_print_string("  PLAYER 2 WINS!");
                }
                
                uint8_t stored_p1 = eeprom_read_byte(&nvHighP1);
                uint8_t stored_p2 = eeprom_read_byte(&nvHighP2);
                if (stored_p1 > 99) stored_p1 = 0;
                if (stored_p2 > 99) stored_p2 = 0;
                
                if ((score_p1 > stored_p1) || (score_p1 == stored_p1 && score_p2 > stored_p2)) {
                    eeprom_update_byte(&nvHighP1, score_p1);
                    eeprom_update_byte(&nvHighP2, score_p2);
                    stored_p1 = score_p1;
                    stored_p2 = score_p2;
                    lcd_send_cmd(0xC0);
                    lcd_print_string(" NEW HIGH SCORE!");
                    _delay_ms(1500);
                    lcd_send_cmd(0x01);
                }
                
                lcd_send_cmd(0xC0);
                lcd_print_string("HiScore: ");
                lcd_send_char((stored_p1 % 10) + '0'); 
                lcd_send_char(':');
                lcd_send_char((stored_p2 % 10) + '0');
                
                lcd_send_cmd(0x01);
                lcd_print_string(" GAME OVER ");
                
                score_p1 = 0;
                score_p2 = 0;
                _delay_ms(3000); 
                
                lcd_hardware_init();
                ball_x = 7;
                ball_y = 0;
                continue;
            }
            
            ball_x = 7; 
            ball_y = 0; 
            velocity_x = (rand() % 2 == 0) ? 1 : -1;
            velocity_y = (rand() % 2 == 0) ? 1 : -1;
            
            continue; 
        }

        if (ball_x == 0 && ball_y == paddle_p1_y)  { velocity_x = 1;  buzzer_beep(5); }
        if (ball_x == 15 && ball_y == paddle_p2_y) { velocity_x = -1; buzzer_beep(5); }

        memset(render_buffer_l1, ' ', 16); 
        memset(render_buffer_l2, ' ', 16);
        render_buffer_l1[16] = render_buffer_l2[16] = '\0';
        
        if (paddle_p1_y == 0) render_buffer_l1[0]  = '|'; else render_buffer_l2[0]  = '|';
        if (paddle_p2_y == 0) render_buffer_l1[15] = '|'; else render_buffer_l2[15] = '|';
        
        if (ball_x >= 0 && ball_x < 16) {
            if (ball_y == 0) render_buffer_l1[ball_x] = 'O'; else render_buffer_l2[ball_x] = 'O';
        }

        lcd_send_cmd(0x80); lcd_print_string(render_buffer_l1);
        lcd_send_cmd(0xC0); lcd_print_string(render_buffer_l2);
        
        for (uint8_t cycles = 0; cycles < 26; cycles++) {
            display_refresh_score(score_p1, score_p2);
        }
    }
}
